"""
Step 1 Agent — Evidence Extraction & Construct Definition Enrichment

Reads each relevant literature source and extracts passages that ground
your core constructs. Produces enriched construct definitions with citations.

CUSTOMIZATION:
  1. Update SYSTEM_PROMPT with your paper's theoretical framework
  2. Update EXTRACTION_PROMPT with your specific constructs
  3. Update the construct source groups in config.py
"""

import json, os
from . import config
from .literature_loader import load_sources, prepare_source_context
from .llm import chat_json, chat

# ── CUSTOMIZE: Replace with YOUR paper's theoretical framework ──────────────
SYSTEM_PROMPT = """\
You are a rigorous research assistant helping build construct definitions \
for an academic paper.

ROLE: You are a critical evidence analyst, NOT a cheerful summariser. Your job \
is to find, weigh, and sometimes REJECT evidence. A passage that merely mentions \
a keyword is NOT evidence for a construct.

RULES:
1. Only cite passages that ACTUALLY APPEAR in the provided sources.
2. Do NOT fabricate, paraphrase beyond the source, or hallucinate quotes.
3. If a source has NO relevant content for a construct, say so explicitly.
4. Apply the EVIDENCE STRENGTH RUBRIC below consistently.
5. Think step-by-step: first identify candidate passages, then evaluate each.

EVIDENCE STRENGTH RUBRIC:
- "strong": Source explicitly defines, operationalises, or empirically tests \
  the concept.
- "moderate": Source discusses the concept substantively but in an adjacent context.
- "weak": Source mentions the concept peripherally, or relevance requires \
  inferential bridging.
"""

EXTRACTION_PROMPT = """\
Read the following literature sources carefully, then extract evidence for \
EACH of the constructs listed below.

CONSTRUCTS TO GROUND:
{constructs}

For each construct, return a JSON array of evidence objects:
{{
  "construct": "Construct Name",
  "sub_dimensions": [
    {{
      "name": "Sub-dimension Name",
      "definition": "Your consolidated definition",
      "evidence": [
        {{
          "source_id": "#01",
          "passage": "Exact quote from the source",
          "page_or_section": "Section/page reference",
          "strength": "strong|moderate|weak",
          "rationale": "Why this passage supports this sub-dimension"
        }}
      ]
    }}
  ]
}}

SOURCES:
{source_context}
"""


def run(constructs: list[str] = None) -> dict:
    """Run Step 1: Evidence Extraction.

    Args:
        constructs: List of construct names to extract evidence for.
                   If None, uses a placeholder list.

    Returns:
        Dict with extraction results.
    """
    constructs = constructs or ["Construct A", "Construct B", "Construct C"]

    print("=" * 60)
    print("STEP 1 — Evidence Extraction")
    print("=" * 60)

    # Load all readable sources
    docs = load_sources(list(config.LITERATURE_FILES.keys()))
    if not docs:
        print("  No literature sources found. Add files to literature/ and")
        print("  configure LITERATURE_FILES in scripts/agents/config.py")
        return {"constructs": constructs, "evidence": [], "docs_loaded": 0}

    print(f"  Loaded {len(docs)} sources")

    # Prepare context
    context = prepare_source_context(docs)

    # Build prompt
    construct_text = "\n".join(f"  - {c}" for c in constructs)
    prompt = EXTRACTION_PROMPT.format(
        constructs=construct_text,
        source_context=context,
    )

    print("  Extracting evidence from literature...")
    result = chat_json(
        messages=[
            {"role": "system", "content": SYSTEM_PROMPT},
            {"role": "user", "content": prompt},
        ],
        model=config.EXTRACTION_MODEL,
    )

    # Save output
    out_path = os.path.join(config.ENRICHED_DIR, "step1_evidence.json")
    with open(out_path, "w", encoding="utf-8") as f:
        json.dump(result, f, indent=2, ensure_ascii=False)
    print(f"  Saved → {out_path}")

    return {"constructs": constructs, "evidence": result, "docs_loaded": len(docs)}
