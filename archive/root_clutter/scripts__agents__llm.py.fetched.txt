"""
LLM interaction utilities — wraps OpenAI client with retry logic,
structured output parsing, and token tracking.
"""

import json, time, re
from openai import OpenAI, RateLimitError, APITimeoutError, APIConnectionError
from . import config

_client = None
_token_usage = {"prompt_tokens": 0, "completion_tokens": 0, "calls": 0}


def get_client() -> OpenAI:
    """Lazy-initialize the OpenAI client."""
    global _client
    if _client is None:
        _client = OpenAI()
    return _client


def chat(messages: list[dict],
         model: str = None,
         temperature: float = None,
         max_tokens: int = None,
         response_format: dict = None) -> str:
    """Send a chat completion request with retry logic.
    Returns the assistant message content as a string."""
    model = model or config.EXTRACTION_MODEL
    temperature = temperature if temperature is not None else config.TEMPERATURE
    max_tokens = max_tokens or config.MAX_OUTPUT_TOKENS
    client = get_client()

    kwargs = {
        "model": model,
        "messages": messages,
        "temperature": temperature,
        "max_tokens": max_tokens,
    }
    if response_format:
        kwargs["response_format"] = response_format

    for attempt in range(config.MAX_RETRIES):
        try:
            resp = client.chat.completions.create(**kwargs)
            _token_usage["prompt_tokens"] += resp.usage.prompt_tokens
            _token_usage["completion_tokens"] += resp.usage.completion_tokens
            _token_usage["calls"] += 1
            return resp.choices[0].message.content
        except (RateLimitError, APITimeoutError, APIConnectionError) as e:
            wait = config.RETRY_DELAY * (attempt + 1)
            print(f"  [RETRY {attempt+1}/{config.MAX_RETRIES}] {type(e).__name__}: waiting {wait}s...")
            time.sleep(wait)
    raise RuntimeError(f"Failed after {config.MAX_RETRIES} retries")


def chat_json(messages: list[dict],
              model: str = None,
              temperature: float = None,
              max_tokens: int = None) -> dict | list:
    """Chat completion that returns parsed JSON."""
    raw = chat(messages, model=model, temperature=temperature,
               max_tokens=max_tokens,
               response_format={"type": "json_object"})
    return json.loads(raw)


def extract_json_from_text(text: str) -> dict | list:
    """Extract JSON from text that might have markdown fences or extra prose."""
    # Direct parse
    try:
        return json.loads(text)
    except json.JSONDecodeError:
        pass
    # Markdown code fence
    match = re.search(r"```(?:json)?\s*\n([\s\S]*?)\n```", text)
    if match:
        try:
            return json.loads(match.group(1))
        except json.JSONDecodeError:
            pass
    # First { or [ to last } or ]
    for start_char, end_char in [("{", "}"), ("[", "]")]:
        start = text.find(start_char)
        end = text.rfind(end_char)
        if start != -1 and end > start:
            try:
                return json.loads(text[start:end+1])
            except json.JSONDecodeError:
                pass
    raise ValueError(f"Could not extract JSON from response: {text[:200]}...")


def get_token_usage() -> dict:
    """Return cumulative token usage statistics."""
    return dict(_token_usage)


def print_usage_summary():
    """Print a summary of total token usage and estimated cost."""
    u = _token_usage
    # Approximate pricing (blended gpt-4.1 + gpt-4.1-mini)
    est_cost = (u["prompt_tokens"] * 1.0 + u["completion_tokens"] * 4.0) / 1_000_000
    print(f"\n{'='*60}")
    print(f"LLM USAGE SUMMARY")
    print(f"{'='*60}")
    print(f"  API calls:         {u['calls']}")
    print(f"  Prompt tokens:     {u['prompt_tokens']:,}")
    print(f"  Completion tokens: {u['completion_tokens']:,}")
    print(f"  Est. cost (blended): ${est_cost:.2f}")
